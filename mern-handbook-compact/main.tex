  morekeywords={
    typeof,new,in,of,instanceof,export,import,from,as,default,extends,super,
    async,await,function,return,this,class,let,const,var,if,else,for,while,
    do,switch,case,break,continue,try,catch,finally,throw,delete,void,yield,
    static,get,set,null,undefined,true,false
  },
  sensitive=true,
  morecomment=[l]{//},
  morecomment=[s]{/*}{*/},
  morestring=[b]',
  morestring=[b]",
  morestring=[b]`
}

  morekeywords={cd,mkdir,npm,npx,node,cp,mv,rm,export,echo,sudo,git},
  sensitive=true,
  morecomment=[l]{\#},
  morestring=[b]",
  morestring=[b]'
}
